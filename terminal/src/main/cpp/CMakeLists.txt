cmake_minimum_required(VERSION 3.22.1)
project(hometty_terminal LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
file(GLOB VTERM_SOURCES CONFIGURE_DEPENDS vendor/libvterm/src/*.c)
add_library(homettyterminal SHARED native.cpp ${VTERM_SOURCES})
target_include_directories(homettyterminal PRIVATE vendor/libvterm/include vendor/libvterm/src)
target_compile_options(homettyterminal PRIVATE -Wall -Wextra)
target_link_options(homettyterminal PRIVATE "-Wl,-z,max-page-size=16384")
add_executable(homettycli bridge.cpp)
set_target_properties(homettycli PROPERTIES OUTPUT_NAME "libhomettycli.so")
target_compile_options(homettycli PRIVATE -Wall -Wextra)
target_link_options(homettycli PRIVATE "-Wl,-z,max-page-size=16384")
